/** Matches SchedulerCtaButton's slow-hint delay without importing that React module. */
const STATIC_SCHEDULER_HINT_DELAY_MS = 500;

/** Present in the bundled module so the GTM check can find this boot file. */
export const SRA_SITE_RUNTIME_MARKER = "sra-site-runtime-boot";

type SchedulerWindow = Window & {
  __sraOpenScheduler?: () => void;
};

function liveRegion(button: Element): HTMLElement | null {
  const next = button.nextElementSibling;
  if (next instanceof HTMLElement && next.getAttribute("aria-live")) return next;
  return null;
}

export function bindStaticSchedulerHints(): void {
  let armedButton: HTMLButtonElement | null = null;
  let hintTimer = 0;

  document.addEventListener(
    "click",
    (event) => {
      const target = event.target;
      if (!(target instanceof Element)) return;
      const button = target.closest<HTMLButtonElement>("[data-sra-scheduler-cta]");
      if (!button || button.hasAttribute("data-sra-scheduler-ready")) return;
      armedButton = button;
      window.clearTimeout(hintTimer);
      hintTimer = window.setTimeout(() => {
        if (armedButton !== button) return;
        if (getServiceTitanSchedulerStatus() !== "opening") return;
        button.setAttribute("aria-busy", "true");
        const live = liveRegion(button);
        if (live) live.textContent = "Opening scheduler…";
      }, STATIC_SCHEDULER_HINT_DELAY_MS);
    },
    true,
  );

  subscribeServiceTitanSchedulerStatus((status) => {
    const button = armedButton;
    if (!button || button.hasAttribute("data-sra-scheduler-ready")) return;
    const live = liveRegion(button);
    if (status === "unavailable") {
      window.clearTimeout(hintTimer);
      button.removeAttribute("aria-busy");
      const label = button.getAttribute("aria-label") || button.textContent || "Schedule";
      if (live) live.textContent = `${label.trim()} unavailable. Tap to retry.`;
      return;
    }
    if (status !== "opening") {
      window.clearTimeout(hintTimer);
      button.removeAttribute("aria-busy");
      if (live) live.textContent = "";
      armedButton = null;
    }
  });
}

function bindStaticPhoneClicks(): void {
  document.addEventListener("click", (event) => {
    const target = event.target;
    if (!(target instanceof Element)) return;
    // Blog article copy keeps the React BlogArticleBody rule: any call link in the
    // body counts as "blog-article-body".
    const link =
      target.closest<HTMLAnchorElement>("a[data-ff-phone-location]") ??
      target.closest<HTMLAnchorElement>('.blog-article-body a[href^="tel:"]');
    if (!link || link.closest("astro-island")) return;
    const attribution = getStoredAttribution();
    const pageContext = document.querySelector("[data-sra-city-slug]");
    pushDataLayerEvent({
      event: "click_to_call",
      cta_location:
        link.getAttribute("data-ff-phone-location") ||
        (link.closest(".blog-article-body") ? "blog-article-body" : undefined),
      page_path: window.location.pathname,
      city_slug:
        link.getAttribute("data-sra-city-slug") ||
        pageContext?.getAttribute("data-sra-city-slug") ||
        null,
      service_slug:
        link.getAttribute("data-sra-service-slug") ||
        pageContext?.getAttribute("data-sra-service-slug") ||
        null,
      ...attributionToEventFields(attribution),
    });
  });
}

function bindStaticFaqAccordions(): void {
  document.addEventListener("click", (event) => {
    const target = event.target;
    if (!(target instanceof Element)) return;
    const button = target.closest<HTMLButtonElement>('button[id^="faq-button-"]');
    if (!button || button.closest("astro-island")) return;
    const panelId = button.getAttribute("aria-controls");
    const panel = panelId ? document.getElementById(panelId) : null;
    if (!panel) return;
    event.preventDefault();
    const expanded = button.getAttribute("aria-expanded") === "true";
    const section = button.closest("section") ?? document;
    for (const other of section.querySelectorAll<HTMLButtonElement>('button[id^="faq-button-"]')) {
      if (other.closest("astro-island")) continue;
      const otherPanelId = other.getAttribute("aria-controls");
      const otherPanel = otherPanelId ? document.getElementById(otherPanelId) : null;
      const open = other === button ? !expanded : false;
      other.setAttribute("aria-expanded", open ? "true" : "false");
      const icon = other.querySelector("[aria-hidden='true']");
      if (icon) icon.textContent = open ? "\u2212" : "+";
      if (otherPanel) {
        if (open) otherPanel.removeAttribute("hidden");
        else otherPanel.setAttribute("hidden", "");
        otherPanel.setAttribute("aria-hidden", open ? "false" : "true");
      }
    }
  });
}

/**
 * Same boot order as the old React island: scheduler hook, attribution,
 * GTM 3.5 s after page load (or first gesture), hydrated signal, then route-commit. No React runtime.
 */
export function bootAstroSiteRuntime(): void {
  const schedulerWindow = window as SchedulerWindow;
  const open = () => openServiceTitanScheduler();
  schedulerWindow.__sraOpenScheduler = open;
  armJourneyListeners();
  consumePreHydrationSchedulerIntent();
  bindStaticSchedulerHints();
  bindStaticPhoneClicks();
  bindStaticFaqAccordions();

  const pathname = window.location.pathname;
  const search = window.location.search;
  const snapshot = captureAttributionOnEntry(pathname, search);
  if (hasPaidClickParams(search)) {
    void collectAttributionVisit(snapshot, pathname);
  }
  startJourneyTracking();
  loadGtm();
  bindHeaderScroll();
  window.__SRA_HYDRATED__ = true;
  window.__SRA_SITE_RUNTIME__ = SRA_SITE_RUNTIME_MARKER;
  window.dispatchEvent(new Event("sra:hydrated"));
  waitForRoutePresentationReady(pathname, () => {
    dispatchRouteCommitted(pathname);
  });
}

declare global {
  interface Window {
    __SRA_SITE_RUNTIME__?: string;
  }
}
