import assert from "node:assert/strict";
import { afterEach, describe, it } from "node:test";
import { Window } from "happy-dom";
import {
  buildMeasurementEvent,
  resetMeasurementForTests,
  sentMeasurementBody,
} from "../../src/lib/measurement/collect.ts";
import { stripUnknownAndPii, validEntry } from "../../src/lib/measurement/contract.ts";
import {
  MEASUREMENT_FORWARD_KEYS,
} from "../../server/lib/forwardMeasurementCollect.ts";

function installWindow(cookies: string[]) {
  const win = new Window({ url: "https://www.sraroofs.com/springfield/" });
  for (const cookie of cookies) win.document.cookie = cookie;
  (globalThis as typeof globalThis & { window: Window }).window = win;
  (globalThis as typeof globalThis & { document: Document }).document =
    win.document as unknown as Document;
  return win;
}

afterEach(() => {
  resetMeasurementForTests();
  delete (globalThis as { window?: unknown }).window;
  delete (globalThis as { document?: unknown }).document;
});

describe("ff_entry entry tag", () => {
  it("validates the value format", () => {
    assert.equal(validEntry("gbp-springfield"), "gbp-springfield");
    assert.equal(validEntry("qr-yard-sign-1"), "qr-yard-sign-1");
    assert.equal(validEntry(" gbp-branson "), "gbp-branson");
    for (const bad of ["", "gbp", "gbp-", "-springfield", "GBP-springfield", "gbp springfield", "gbp-" + "a".repeat(61)]) {
      assert.equal(validEntry(bad), null, bad);
    }
    assert.equal(validEntry(null), null);
  });

  it("sends the ff_entry cookie as entry on every event", () => {
    installWindow(["ff_entry=gbp-springfield"]);
    for (const name of ["page_view", "click_to_call", "lead_form_success", "scheduler_opened"] as const) {
      const body = sentMeasurementBody(buildMeasurementEvent(name));
      assert.equal(body.entry, "gbp-springfield", name);
    }
  });

  it("omits entry without the cookie or with a malformed one", () => {
    installWindow([]);
    assert.equal("entry" in sentMeasurementBody(buildMeasurementEvent("page_view")), false);
    resetMeasurementForTests();
    installWindow(["ff_entry=Not Valid"]);
    assert.equal("entry" in sentMeasurementBody(buildMeasurementEvent("page_view")), false);
  });

  it("keeps entry through the client sanitizer and the forwarder allowlist", () => {
    assert.equal(stripUnknownAndPii({ entry: "gbp-springfield" }).entry, "gbp-springfield");
    assert.ok((MEASUREMENT_FORWARD_KEYS as readonly string[]).includes("entry"));
  });
});
