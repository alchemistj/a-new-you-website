import { SITE_ORIGIN } from "@/lib/siteImages";

/** Dedicated Fluid Frame Forms public intake. Not a per-site Edge Function. */
export const FORMS_PUBLIC_ORIGIN = "https://forms.fluidframemarketing.com";
export const FORMS_LEAD_ENDPOINT = "https://forms.fluidframemarketing.com/api/lead";
/** Same-origin Preview-only forwarder. Production browsers keep using FORMS_LEAD_ENDPOINT. */
export const PREVIEW_FORMS_LEAD_PATH = "/api/preview-forms-lead";

/** Stable catalog keys unless Forms later documents a different canonical value. */
export const FORMS_CLIENT_KEY = "greene-planet";
export const FORMS_SITE_KEY = "greene-planet-website";
export const FORMS_FORM_KEY = "main-contact";

export type FormsCustomField = {
  label: string;
  value: string;
  type: string;
};

export type FormsLeadRequest = {
  siteKey: string;
  formKey: string;
  clientSubmissionId: string;
  name: string;
  email: string;
  phone: string;
  message: string;
  customFields: Record<string, FormsCustomField>;
  rawPayload: Record<string, unknown>;
  pageUrl: string;
  referrer: string | null;
  utm: {
    source: string;
    medium: string;
    campaign: string;
    term: string;
    content: string;
  };
  honeypot: string;
  formStartedAt: string;
  submittedAt: string;
};

export type FormsLeadResponse = {
  ok?: boolean;
  conversionEligible?: boolean;
  duplicate?: boolean;
  id?: string;
  reason?: "validation" | "network";
  message?: string;
  error?: string;
  code?: string;
};

export function customField(
  label: string,
  value: string | boolean | number | null | undefined,
  type = "text",
): FormsCustomField | undefined {
  if (value === null || value === undefined) return undefined;
  if (typeof value === "string" && !value.trim()) return undefined;
  return { label, value: String(value), type };
}

export function addCustomField(
  fields: Record<string, FormsCustomField>,
  key: string,
  field: FormsCustomField | undefined,
) {
  if (field) fields[key] = field;
}

export function resolveFormsPageUrl(sourcePage: string, pagePath?: string | null): string {
  if (typeof window !== "undefined" && window.location?.href) {
    return window.location.href;
  }
  const candidate = (pagePath || sourcePage).trim();
  if (/^https?:\/\//i.test(candidate)) return candidate;
  const path = candidate.startsWith("/") ? candidate : `/${candidate}`;
  return `${SITE_ORIGIN}${path}`;
}

export function isVercelPreviewHost(hostname: string): boolean {
  const host = hostname.trim().toLowerCase().split(":")[0] ?? "";
  return host.endsWith(".vercel.app");
}

export function getFormsLeadEndpoint(hostname?: string): string {
  const host =
    hostname ??
    (typeof window !== "undefined" && window.location?.hostname ? window.location.hostname : "");
  if (isVercelPreviewHost(host)) return PREVIEW_FORMS_LEAD_PATH;
  return FORMS_LEAD_ENDPOINT;
}

export function getFormsPublicOrigin(): string {
  return FORMS_PUBLIC_ORIGIN;
}

export function mapFormsLeadResponse(payload: FormsLeadResponse | null): {
  accepted: boolean;
  acceptedId?: string;
} {
  const acceptedId =
    payload?.conversionEligible === true && typeof payload.id === "string" ? payload.id.trim() : "";
  if (acceptedId) {
    return { accepted: true, acceptedId };
  }
  return { accepted: false };
}
